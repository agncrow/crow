handler board_cards_make_board (board/cards.co)
  wired: board_cards_boards_view button 'Make'
  when (__reader[].role neq 'admin'):
    effect notice 'only an admin makes boards'
  when (board_cards_new_board eq '') and (not (__reader[].role neq 'admin')):
    nothing
  when (not (__reader[].role neq 'admin')) and (not (board_cards_new_board eq '')):
    board_cards_new_board := ''
    effect query board_cards_book 'INSERT INTO boards (name) VALUES (?)' [board_cards_new_board]
    calls board_store_note[board_cards_book, __reader[].name, 'made the board `board_cards_new_board`']
