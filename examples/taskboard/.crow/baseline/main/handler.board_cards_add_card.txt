handler board_cards_add_card (board/cards.co)
  wired: board_cards_board_view button 'Add'
  when (board_cards_new_card eq ''):
    nothing
  when (not (board_cards_new_card eq '')):
    board_cards_new_card := ''
    effect query board_cards_book 'INSERT INTO cards (board, title, status, owner, stars) VALUES (?, ?, ?, ?, 0)' [board_cards_board_id, board_cards_new_card, 'todo', __reader[].name]
    calls board_store_note[board_cards_book, __reader[].name, 'added `board_cards_new_card`']
