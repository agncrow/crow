handler board·cards·advance (board/cards.co)
  inputs: $1
  wired: board·cards·board_view table action 'Advance'
  reads v1 = query[board·cards·book, 'SELECT title, status FROM cards WHERE id = ?', [$1]]
  when (len[v1] eq 0):
    nothing
  when (not (len[v1] eq 0)) and (not (v1.status[0] eq 'todo')):
    effect query board·cards·book 'UPDATE cards SET status = ? WHERE id = ?' ['done', $1]
    calls board·store·note[board·cards·book, __reader[].name, 'moved `v1.title[0]` to `'done'`']
  when (not (len[v1] eq 0)) and (v1.status[0] eq 'todo'):
    effect query board·cards·book 'UPDATE cards SET status = ? WHERE id = ?' ['doing', $1]
    calls board·store·note[board·cards·book, __reader[].name, 'moved `v1.title[0]` to `'doing'`']
