
#[derive(Debug, Clone)]
struct Tree {
    left: Option<Box<Tree>>,
    right: Option<Box<Tree>>,
}

fn check_tree(tree: &Tree) -> i64 {
    if tree.left.is_none() {
        return 1_i64;
    }
    return 1_i64 + check_tree(tree.left.as_ref().unwrap()) + check_tree(tree.right.as_ref().unwrap());
}

fn make_tree(depth: i64) -> Tree {
    if depth == 0_i64 {
        return Tree { left: None, right: None };
    }
    return Tree { left: Some(Box::new(make_tree(depth - 1_i64))), right: Some(Box::new(make_tree(depth - 1_i64))) };
}

fn main() {
    let min_depth: i64 = 4_i64;
    let max_depth: i64 = 15_i64;
    let stretch_depth: i64 = max_depth + 1_i64;
    let stretch_tree: Tree = make_tree(stretch_depth);
    println!("stretch tree of depth {} check: {}", stretch_depth, check_tree(&stretch_tree));
    let long_lived_tree: Tree = make_tree(max_depth);
    let mut depth: i64 = min_depth;
    while depth <= max_depth {
        let mut iterations: i64 = 1_i64;
        let mut i: i64 = 0_i64;
        while i < max_depth - depth + min_depth {
            iterations = iterations * 2_i64;
            i = i + 1_i64;
        }
        let mut check_sum: i64 = 0_i64;
        let mut j: i64 = 0_i64;
        while j < iterations {
            check_sum = check_sum + check_tree(&make_tree(depth));
            j = j + 1_i64;
        }
        println!("{} trees of depth {} check: {}", iterations, depth, check_sum);
        depth = depth + 2_i64;
    }
    println!("long lived tree of depth {} check: {}", max_depth, check_tree(&long_lived_tree));
}
